        "\n",
        "import re\n",
        "import numpy as np\n",
        "from sklearn.feature_extraction.text import TfidfVectorizer\n",
        "from sklearn.metrics.pairwise import cosine_similarity\n",
        "\n",
        "\n",
        "# ============================================================\n",
        "# 1. DAY 10 PIPELINE\n",
        "# ============================================================\n",
        "\n",
        "class Pipeline:\n",
        "    def __init__(self):\n",
        "        self.vectorizer = TfidfVectorizer(\n",
        "            lowercase=True,\n",
        "            stop_words=\"english\"\n",
        "        )\n",
        "\n",
        "    def preprocess(self, text):\n",
        "        text = text.lower()\n",
        "        text = re.sub(r\"[^a-zA-Z0-9\\s]\", \"\", text)\n",
        "        text = re.sub(r\"\\s+\", \" \", text).strip()\n",
        "        return text\n",
        "\n",
        "    def fit_transform(self, documents):\n",
        "        cleaned_docs = [self.preprocess(doc) for doc in documents]\n",
        "        return self.vectorizer.fit_transform(cleaned_docs)\n",
        "\n",
        "    def transform(self, text):\n",
        "        cleaned_text = self.preprocess(text)\n",
        "        return self.vectorizer.transform([cleaned_text])\n",
        "\n",
        "\n",
        "# ============================================================\n",
        "# 2. KNOWLEDGE BASE\n",
        "# ============================================================\n",
        "\n",

        "]\n",
        "\n",
        "\n",
        "# ============================================================\n",
        "# 3. CREATE PIPELINE AND TF-IDF MATRIX\n",
        "# ============================================================\n",
        "\n",
        "pipeline = Pipeline()\n",
        "\n",
        "corpus_matrix = pipeline.fit_transform(documents)\n",
        "\n",
        "print(\"Number of documents:\", len(documents))\n",
        "print(\"TF-IDF matrix shape:\", corpus_matrix.shape)\n",
        "\n",
        "\n",
        "# ============================================================\n",
        "# 4. RETRIEVAL FUNCTION\n",
        "# ============================================================\n",
        "\n",
        "def retrieve(query, corpus_matrix, top_k=3, threshold=0.1):\n",
        "\n",
        "    # Vectorise query\n",
        "    query_vector = pipeline.transform(query)\n",
        "\n",
        "    # Calculate cosine similarity\n",
        "    similarities = cosine_similarity(\n",
        "        query_vector,\n",
        "        corpus_matrix\n",
        "    )[0]\n",
        "\n",
        "    # Highest similarity\n",
        "    max_score = np.max(similarities)\n",
        "\n",
        "    # Relevance threshold\n",
        "    if max_score < threshold:\n",
        "        return \"No relevant document found\"\n",
        "\n",
        "    # Get top K document indexes\n",
        "    top_indices = np.argsort(similarities)[::-1][:top_k]\n",
        "\n",
        "    results = []\n",
        "\n",
        "    for index in top_indices:\n",
        "        results.append({\n",
        "            \"document_id\": index + 1,\n",
        "            \"document\": documents[index],\n",
        "            \"similarity_score\": round(float(similarities[index]), 4)\n",
        "        })\n",
        "\n",
        "    return results\n",
        "\n",
        "\n",
